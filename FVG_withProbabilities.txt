isDisp(ofs, wantBull) =>
    bodyAbs = math.abs(close[ofs] - open[ofs])
    rng     = math.max(high[ofs] - low[ofs], syminfo.mintick)
    atrRef  = math.max(atr, syminfo.mintick)
    big     = bodyAbs / atrRef >= i_dispBodyAtr
    solid   = bodyAbs / rng    >= i_dispBodyRat
    dir     = wantBull ? close[ofs] > open[ofs] : close[ofs] < open[ofs]
    big and solid and dir

clamp(v, lo, hi) => math.max(lo, math.min(hi, v))
logit(p) => math.log(p / (1.0 - p))

calcScore(fvgSize, fvgMid, isBull) =>
    sizeRatio = fvgSize / math.max(atr, syminfo.mintick)
    f1 = sizeRatio > 1.5 ? 0.82 : sizeRatio > 0.9 ? 0.65 : sizeRatio > 0.4 ? 0.50 : 0.35
    f2 = isBull ? (bullTrend ? 0.78 : 0.30) : (bearTrend ? 0.78 : 0.30)
    f3 = isBull ? (aboveHTF ? 0.76 : 0.34) : (belowHTF ? 0.76 : 0.34)
    f4 = volRelDisp > 1.5 ? 0.72 : volRelDisp > 0.8 ? 0.55 : 0.38
    f5 = not haveRange ? 0.50 : (isBull ? (fvgMid < rangeMid ? 0.80 : 0.30) : (fvgMid > rangeMid ? 0.80 : 0.30))
    f6 = isBull ? (bosBull ? 0.75 : 0.40) : (bosBear ? 0.75 : 0.40)
    f7 = inKZ ? 0.68 : 0.45
    f8 = volNormal ? 0.60 : volExpanded ? 0.75 : 0.35
    ll = 1.2*logit(clamp(f1,0.1,0.9)) + 1.0*logit(clamp(f2,0.1,0.9)) + 1.0*logit(clamp(f3,0.1,0.9)) + 0.8*logit(clamp(f4,0.1,0.9)) + 1.5*logit(clamp(f5,0.1,0.9)) + 1.3*logit(clamp(f6,0.1,0.9)) + 0.7*logit(clamp(f7,0.1,0.9)) + 0.9*logit(clamp(f8,0.1,0.9))
    clamp(1.0 / (1.0 + math.exp(-(-0.60 + ll))), 0.04, i_maxScore)

probColor(p, isBull) =>
    tier = p >= i_highProb ? 2 : p >= i_medProb ? 1 : 0
    isBull ? (tier == 2 ? i_bullHigh : tier == 1 ? i_bullMed : i_bullLow) : (tier == 2 ? i_bearHigh : tier == 1 ? i_bearMed : i_bearLow)

tierLabel(p) => p >= i_highProb ? "H" : p >= i_medProb ? "M" : "L"


var array<box>   fvg_boxes    = array.new<box>()
var array<line>  fvg_ceLns    = array.new<line>()

var array<float> fvg_probs    = array.new<float>()
var array<int>   fvg_state    = array.new<int>()
var array<int>   fvg_birthBar = array.new<int>()

overlaps(newTop, newBot) =>
    bool hit = false
    sz = array.size(fvg_boxes)
    if sz > 0
        for i = 0 to sz - 1
            exTop = array.get(fvg_tops, i)
            exBot = array.get(fvg_bots, i)
            if (newTop <= exTop and newTop >= exBot) or (newBot <= exTop and newBot >= exBot) or (newBot <= exBot and newTop >= exTop) or (exBot <= newBot and exTop >= newTop)
                hit := true
                break
    hit

// Detection
bullGap = low - high[2]
bearGap = low[2] - high
bullFVG = low > high[2] and bullGap >= i_minAtrMult * atr and (not i_useDisp or c2_bullDisp)
bearFVG = high < low[2] and bearGap >= i_minAtrMult * atr and (not i_useDisp or c2_bearDisp)

if bullFVG and not overlaps(low, high[2])
    mid = (low + high[2]) / 2
    p   = calcScore(bullGap, mid, true)
    c   = probColor(p, true)
    lbl = tierLabel(p) + " " + str.format("{0,number,#}%", p * 100)
    bx  = box.new(bar_index[2], low, bar_index + i_extendBars, high[2], bgcolor=color.new(c, i_fillAlpha), border_color=color.new(c, i_borderAlpha), border_width=1, extend=extend.right, text=lbl, text_color=color.new(c, 10), text_size=size.normal, text_halign=text.align_right, text_valign=text.align_top)
    ceLn = i_showCE ? line.new(bar_index[2], mid, bar_index + i_extendBars, mid, color=color.new(c, 30), style=line.style_dotted, extend=extend.right) : na
    array.push(fvg_boxes, bx)
    array.push(fvg_ceLns, ceLn)
    array.push(fvg_tops, low)
    array.push(fvg_bots, high[2])
    array.push(fvg_isBull, true)
    array.push(fvg_probs, p)
    array.push(fvg_state, 0)
    array.push(fvg_birthBar, bar_index)

if bearFVG and not overlaps(low[2], high)
    mid = (low[2] + high) / 2
    p   = calcScore(bearGap, mid, false)
    c   = probColor(p, false)
    lbl = tierLabel(p) + " " + str.format("{0,number,#}%", p * 100)
    bx  = box.new(bar_index[2], low[2], bar_index + i_extendBars, high, bgcolor=color.new(c, i_fillAlpha), border_color=color.new(c, i_borderAlpha), border_width=1, extend=extend.right, text=lbl, text_color=color.new(c, 10), text_size=size.normal, text_halign=text.align_right, text_valign=text.align_top)
    ceLn = i_showCE ? line.new(bar_index[2], mid, bar_index + i_extendBars, mid, color=color.new(c, 30), style=line.style_dotted, extend=extend.right) : na
    array.push(fvg_boxes, bx)
    array.push(fvg_ceLns, ceLn)
    array.push(fvg_tops, low[2])
    array.push(fvg_bots, high)
    array.push(fvg_isBull, false)
    array.push(fvg_probs, p)
    array.push(fvg_state, 0)
    array.push(fvg_birthBar, bar_index)


atrDist = i_ifvgCloseDist * atr
var array<int> toDelete = array.new<int>()
array.clear(toDelete)

// Track whether CE was tapped this bar 
bool ceTapBullFVG = false
bool ceTapBearFVG = false
float ceTapBullScore = 0.0
float ceTapBearScore = 0.0

sz = array.size(fvg_boxes)
if sz > 0
    for i = 0 to sz - 1
        t     = array.get(fvg_tops,     i)
        b     = array.get(fvg_bots,     i)
        bull  = array.get(fvg_isBull,   i)
        st    = array.get(fvg_state,    i)
        pSc   = array.get(fvg_probs,    i)
        birth = array.get(fvg_birthBar, i)
        mid   = (t + b) / 2
        bx    = array.get(fvg_boxes,    i)
        age   = bar_index - birth

        if st == 0
            ceTap = bull ? low <= mid : high >= mid
            if ceTap
                array.set(fvg_state, i, 1)
                box.set_border_width(bx, 2)
                if bull and pSc > ceTapBullScore
                    ceTapBullFVG := true
                    ceTapBullScore := pSc
                if not bull and pSc > ceTapBearScore
                    ceTapBearFVG := true
                    ceTapBearScore := pSc

        if age > i_ageHard
            array.push(toDelete, i)
        else if age > i_ageSoft
            curBull = array.get(fvg_isBull, i)
            newC = st == 3 ? (curBull ? i_ifvgBull : i_ifvgBear) : probColor(pSc, curBull)
            box.set_bgcolor(bx, color.new(newC, math.min(95, i_fillAlpha + i_dimExtra)))
            box.set_border_color(bx, color.new(newC, math.min(95, i_borderAlpha + i_dimExtra + 10)))

        if st != 3
            fullFill = bull ? low <= b : high >= t
            if fullFill
                closeBeyond = bull ? close < b : close > t
                closeDistOk = bull ? (b - close) >= atrDist : (close - t) >= atrDist
                dispOk      = not i_ifvgReqDisp or (bull ? curBearDisp : curBullDisp)
                bosOk       = not i_ifvgReqBOS  or (bull ? bosBear     : bosBull)
                kzOk        = not i_ifvgKzOnly  or inKZ
                validInv    = i_trackIFVG and closeBeyond and closeDistOk and dispOk and bosOk and kzOk

                if validInv
                    newC = bull ? i_ifvgBull : i_ifvgBear
                    newTop = t
                    newBot = b
                    if i_ifvgRedraw
                        newTop := math.max(open, close)
                        newBot := math.min(open, close)
                        array.set(fvg_tops, i, newTop)
                        array.set(fvg_bots, i, newBot)
                        box.set_top(bx, newTop)
                        box.set_bottom(bx, newBot)
                        newMid = (newTop + newBot) / 2
                        ceLn = array.get(fvg_ceLns, i)
                        if not na(ceLn)
                            line.set_y1(ceLn, newMid)
                            line.set_y2(ceLn, newMid)
                            line.set_color(ceLn, color.new(newC, 30))
                    box.set_bgcolor(bx, color.new(newC, i_fillAlpha - 5))
                    box.set_border_color(bx, color.new(newC, i_borderAlpha))
                    box.set_text(bx, "IFVG " + str.format("{0,number,#}%", pSc * 100))
                    array.set(fvg_isBull, i, not bull)
                    array.set(fvg_state,  i, 3)
                else
                    shouldDel = i_fullMitigate ? fullFill : (bull ? low <= t : high >= b)
                    if shouldDel
                        array.push(toDelete, i)

        if st == 3
            tNow    = array.get(fvg_tops,   i)
            bNow    = array.get(fvg_bots,   i)
            nowBull = array.get(fvg_isBull, i)
            broken  = nowBull ? close < bNow : close > tNow
            if broken
                array.push(toDelete, i)

// Distance hiding
if i_enableDist and barstate.islast
    sz3 = array.size(fvg_boxes)
    if sz3 > 0
        for i = 0 to sz3 - 1
            t2 = array.get(fvg_tops, i)
            b2 = array.get(fvg_bots, i)
            m2 = (t2 + b2) / 2
            dist = math.abs(close - m2) / math.max(atr, syminfo.mintick)
            bx2 = array.get(fvg_boxes, i)
            if dist > i_distMax
                box.set_bgcolor(bx2, color.new(color.gray, 99))
                box.set_border_color(bx2, color.new(color.gray, 99))
                box.set_text(bx2, "")
                ceLn2 = array.get(fvg_ceLns, i)
                if not na(ceLn2)
                    line.set_color(ceLn2, color.new(color.gray, 99))

// Deletions
delSz = array.size(toDelete)
if delSz > 0
    var array<int> seen = array.new<int>()
    array.clear(seen)
    for j = delSz - 1 to 0
        idx = array.get(toDelete, j)
        if not array.includes(seen, idx) and idx < array.size(fvg_boxes)
            array.push(seen, idx)
            box.delete(array.get(fvg_boxes, idx))
            ceLn3 = array.get(fvg_ceLns, idx)
            if not na(ceLn3)
                line.delete(ceLn3)
            array.remove(fvg_boxes,    idx)
            array.remove(fvg_ceLns,    idx)
            array.remove(fvg_tops,     idx)
            array.remove(fvg_bots,     idx)
            array.remove(fvg_isBull,   idx)
            array.remove(fvg_probs,    idx)
            array.remove(fvg_state,    idx)
            array.remove(fvg_birthBar, idx)

tierOf(isLong, isTrend) =>
    triggerCount = 0
    if isLong
        triggerCount += (trig_longCE     ? 1 : 0)
        triggerCount += (trig_longOB     ? 1 : 0)
        triggerCount += (trig_longSweep  ? 1 : 0)
    else
        triggerCount += (trig_shortCE    ? 1 : 0)
        triggerCount += (trig_shortOB    ? 1 : 0)
        triggerCount += (trig_shortSweep ? 1 : 0)
    velOk = isLong ? bullVelocity >= 2 : bearVelocity >= 2
    highScore = isLong ? trigScoreLong >= i_highProb : trigScoreShort >= i_highProb
    if isTrend and triggerCount >= 2 and volExpanded and velOk
        "A+"
    else if isTrend and (triggerCount >= 2 or (highScore and velOk))
        "A"
    else
        "B"

avgProb(wantBull) =>
    float s = 0.0
    float w = 0.0
    n = array.size(fvg_probs)
    if n > 0
        for i = 0 to n - 1
            if array.get(fvg_isBull, i) == wantBull and array.get(fvg_state, i) != 3
                m    = (array.get(fvg_tops, i) + array.get(fvg_bots, i)) / 2
                dist = math.abs(close - m) / math.max(atr, syminfo.mintick)
                wt   = 1.0 / (1.0 + dist)
                s += array.get(fvg_probs, i) * wt
                w += wt
    w > 0 ? s / w : na
